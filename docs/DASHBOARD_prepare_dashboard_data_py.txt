# ENCODAGE ONE-HOT
# =========================================

def one_hot_encoder(df, nan_as_category=True):
    """
    Encode les variables catégorielles en variables numériques 0/1.

    Objectif :
    - reproduire la logique du notebook de modélisation ;
    - obtenir des colonnes compatibles avec LightGBM et l'API ;
    - éviter les valeurs booléennes dans le payload API.
    """

    categorical_columns = [
        col
        for col in df.columns
        if df[col].dtype == "object"
    ]

    df = pd.get_dummies(
        df,
        columns=categorical_columns,
        dummy_na=nan_as_category,
        dtype=int
    )

    return df
